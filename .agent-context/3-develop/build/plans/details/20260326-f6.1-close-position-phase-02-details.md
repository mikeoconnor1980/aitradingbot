### Task 2.2: Add `onClosePosition` handler with confirmation dialog, API call, toasts, and refresh {#task-22-add-oncloseposition-handler}

Implement the full close position flow: open confirmation dialog → on confirm → set loading → optimistic position removal → call `OrderService.placeOrder()` with opposite-side market order → success/error toast → refresh.

- **Complexity**: Medium
- **Risk Factors**: Side mapping (`Long → "sell"`, `Short → "buy"`) must be correct; `PlaceOrderRequest.side` expects lowercase `"buy"` | `"sell"` while `Position.side` is title-case `"Long"` | `"Short"`
- **Files**:
  - `frontend/trading-ui/src/app/features/dashboard/dashboard.component.ts` — Add `onClosePosition()` method
- **Success**:
  - Confirmation dialog shows with order summary (side, type, asset, size)
  - Cancelling dialog takes no action
  - Confirming sets row loading, removes position optimistically, calls `placeOrder`
  - On success: clears loading, shows success toast, triggers refresh
  - On error: restores position, clears loading, shows error toast
- **Dependencies**: Task 2.1 (ViewChild and pending set must exist)

#### Implementation Details

```typescript
// frontend/trading-ui/src/app/features/dashboard/dashboard.component.ts — modification
// Add import for PlaceOrderRequest at top:
import { PlaceOrderRequest } from "../../core/models/place-order.model";

// Add method after onModifyOrder():

public onClosePosition(position: Position): void {
  const positionKey = this.positionsTable?.getPositionKey(position) ?? position.asset + position.side;

  if (this.positionsTable?.loadingPositionKeys.has(positionKey)) {
    return;
  }

  const closeSide: "buy" | "sell" = position.side === "Long" ? "sell" : "buy";

  this._dialog.open(ConfirmDialogComponent, {
    data: {
      title: "Close Position",
      message: `Close ${position.side} ${position.asset}-PERP position?`,
      side: closeSide,
      orderType: "market" as const,
      asset: position.asset + "-PERP",
      size: Math.abs(position.size),
      confirmText: "Close Position",
      cancelText: "Keep Position"
    },
    width: "400px"
  }).afterClosed().subscribe((confirmed: boolean) => {
    if (!confirmed) {
      return;
    }

    const positionIndex = this.positions.findIndex(
      (p) => p.asset === position.asset && p.side === position.side
    );
    if (positionIndex < 0) {
      return;
    }

    const removedPosition = this.positions[positionIndex];
    this.positionsTable?.setLoading(positionKey, true);
    this._pendingPositionKeys.add(positionKey);
    this.positions = this.positions.filter(
      (p) => !(p.asset === position.asset && p.side === position.side)
    );

    const closeRequest: PlaceOrderRequest = {
      asset: position.asset,
      side: closeSide,
      orderType: "market",
      price: null,
      size: Math.abs(position.size)
    };

    this._orderService.placeOrder(closeRequest).subscribe({
      next: () => {
        this._pendingPositionKeys.delete(positionKey);
        this.positionsTable?.setLoading(positionKey, false);
        this._snackBar.open("Position close order submitted", "Dismiss", { duration: 3000 });
        this._refresh$.next();
      },
      error: (errorResponse: HttpErrorResponse) => {
        this.positions = [
          ...this.positions.slice(0, positionIndex),
          removedPosition,
          ...this.positions.slice(positionIndex)
        ];
        this._pendingPositionKeys.delete(positionKey);
        this.positionsTable?.setLoading(positionKey, false);
        this._snackBar.open(
          `Failed to close position: ${this._formatErrorPayload(errorResponse)}`,
          "Dismiss",
          { duration: 5000 }
        );
      }
    });
  });
}
```
